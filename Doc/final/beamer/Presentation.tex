\section{Motivation}

\begin{frame}{The problem autoregressive models cannot solve natively}
  \begin{itemize}
    \item Autoregressive generation: one token at a time, left to right.
    \item A token is fixed the moment it is emitted.\\
          \emph{No mechanism to revise an earlier token in light of a later one.}
    \item Global properties (sentiment, factual constraint) live in the \emph{whole} sequence.
  \end{itemize}
  \vspace{0.6em}
  \begin{center}
  \begin{tikzpicture}[font=\footnotesize]
    \foreach \i/\w in {0/The, 1/movie, 2/was, 3/absolutely} {
      \node[box, minimum width=1.4cm] (t\i) at (\i*1.7,0) {\w};
    }
    \node[draw=imsred, thick, rounded corners, minimum width=1.4cm, fill=imsred!10] (t4) at (4*1.7,0) {?};
    \foreach \i in {0,...,3}{ \pgfmathtruncatemacro\j{\i+1} \draw[ar] (t\i) -- (t\j); }
    \draw[imsred, thick, -{Latex}, dashed] (t4.north) to[bend right=35] node[above, font=\scriptsize]{cannot flow back} (t0.north);
  \end{tikzpicture}
  \end{center}
  \begin{center}\small Control means steering the \emph{completed} object. That is a global problem.\end{center}
\end{frame}

\begin{frame}{The energy-based premise, and its one assumption}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      Treat the frozen likelihood as an \textbf{energy} over sequences and sample from it:
      \[ p(\bm{x}) \propto \exp\!\big(-E(\bm{x})\big),\quad E = -\log p_{\text{LM}} + \lambda\,C. \]
      \begin{itemize}
        \item Add any constraint $C$ at inference time.
        \item One frozen model, any objective. \textbf{Plug and play.}
      \end{itemize}
      \vspace{0.4em}
      \structure{The untested assumption:}\\
      \emph{the gradient of the frozen likelihood is a usable search direction on discrete text.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.9]
        \draw[->,gray] (-0.2,0) -- (4.4,0) node[right,font=\scriptsize]{sequence};

      \end{center}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{The machinery: Langevin dynamics}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      {\small Sample $p(\bm{s}) \propto e^{-E(\bm{s})}$ with a gradient-guided chain:}
      \[ \bm{s}_{t+1} = \bm{s}_t + \tfrac{\epsilon_t}{2}\,\nabla_{\bm{s}}\log p(\bm{s}_t) + \sqrt{\epsilon_t}\,\bm{\xi}_t \]
      \vspace{-1.0em}
      \begin{itemize}\small
        \item \textbf{Drift}: uphill.
        \item \textbf{Diffusion}: a \emph{sample}, not a mode.
        \item Needs a \textbf{Lipschitz} drift.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1cm, y=1cm, font=\scriptsize]
        \draw[imsblue!25, thick] plot[domain=0:4.3, samples=60] (\x, {0.55*sin(1.6*\x r)+0.55});
        \foreach \i/\xp in {1/0.35, 2/1.15, 3/1.95, 4/2.75, 5/3.6}

          \draw[-{Latex[length=1.5mm]}, imsblue, thick]

        \node[imsred, anchor=west, font=\tiny] at (0.4,1.75) {small steps, smooth surface};
      \end{tikzpicture}
      \end{center}
      \vspace{0.5em}
      {\small Text is \emph{discrete}: there is no small step.}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{center}\small
  Every method here \structure{forces this update onto discrete tokens}.
  \end{center}
\end{frame}

\begin{frame}{Research questions}
  \vspace{0.3em}
  \begin{enumerate}\small
    \item[RQ1] Does the \emph{input-embedding} gradient give a usable proposal direction? If not, what does?
    \item[RQ2] What does the Metropolis--Hastings correction do in each sampler?
    \item[RQ3a] Does a GFlowNet on this reward learn a high-reward policy?
    \item[RQ3b] Does its tuned energy become more navigable?
    \item[E] (extension) Can an additive constraint steer generation here?
  \end{enumerate}
  \vspace{1.0em}
  \begin{center}
  \begin{minipage}{0.88\textwidth}\small
  \structure{Two ways out, if the gradient fails.} Repair the search (RQ1, RQ2), or abandon the
  search and \emph{train a policy} to absorb it (RQ3). The talk takes them in that order.
  \end{minipage}
  \end{center}
\end{frame}

\begin{frame}{Three candidate explanations}
  \vspace{0.2em}
  {\small Suppose the gradient does not help. \emph{Three} things could be at fault.}
  \vspace{0.5em}
  \begin{center}
  \scriptsize
  \setlength{\tabcolsep}{4pt}

  \begin{tabular}{lp{4.7cm}p{3.5cm}}
    \toprule
    & The claim & Decided by \\
    \midrule
    \structure{\textbf{A}} \textbf{target}
      & No local search could profit from this likelihood. Then the proposal is beside the point.
      & search the same energy \emph{without} a gradient \\
    \structure{\textbf{B}} \textbf{model}
      & Teacher forcing only ever asks for the next token, never to \emph{rank a substitution}. Then only retraining repairs it.

    \structure{\textbf{C}} \textbf{coordinates}

  \end{tabular}
  \end{center}

  \structure{Everything that follows is that elimination, in that order.}
  \end{center}

\section{The faithful implementation}

\begin{frame}{Two samplers, implemented faithfully}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\small
        \item \textbf{DLS} (discrete): stays in token space.
        \item \textbf{CLS} (continuous): relaxes into embedding space, projects back.
        \item \textbf{Metropolis--Hastings} correction, applied exactly.
        \item Step size by \textbf{oracle calibration}.
      \end{itemize}
      \vspace{0.3em}
      {\scriptsize DLS scores a candidate $v$ against $x_i$, with $\bm{g} = \nabla_{\bm{e}_i}\log p_{\text{LM}}$:}
      {\footnotesize
      \begin{align*}
        \log q(v \mid x_i) \propto\;\; & \underbrace{\tfrac{1}{2}\bm{g}^\top\!\big(\bm{e}(v)-\bm{e}(x_i)\big)}_{\text{gradient alignment}} \\[0.2em]
        -\;\; & \underbrace{\tfrac{1}{2\alpha}\big\lVert \bm{e}(v)-\bm{e}(x_i)\big\rVert^2}_{\text{distance}}
      \end{align*}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=6mm and 9mm]
        \node[box] (s) {state};
        \node[op, right=of s] (g) {gradient\\$\to$ proposal};
        \node[op, below=of g] (mh) {MH\\accept / reject};
        \node[box, left=of mh] (p) {project\\to token};
        \draw[ar] (s) -- (g);
        \draw[ar] (g) -- (mh);
        \draw[ar] (mh) -- (p);
        \draw[ar] (p) -- (s);
        \node[font=\scriptsize, imsred, below=1.5mm of mh, text width=3.2cm, align=center] {the step the field skips};
      \end{tikzpicture}
      \\[0.4em]{\scriptsize A fixed annealed schedule, never stopped early.}
      \end{center}
      \vspace{0.8em}
      {\small This $\bm{g}$, the \structure{input-embedding gradient}, is the object under test.}
    \end{column}
  \end{columns}
\end{frame}

\section{The null}

\begin{frame}{First: what is the proposal we are about to ablate?}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \small
      \structure{Entropy} measures how spread out a distribution is, in \emph{nats}.
      \begin{itemize}\small
        \item Concentrated on one token: $0$.
        \item Flat over all $50{,}257$: $\log|V| = 10.825$, the ceiling.
      \end{itemize}
      \vspace{0.3em}
      {\small So entropy answers: \emph{how many tokens is the proposal really choosing between?}}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{block}{The calibrated proposal, measured}
        \small
        at its \emph{sharpest}: $10.8248$ nats\\
        uniform ceiling: $10.8249$ nats
      \end{block}
      \vspace{-0.2em}
      {\small Effectively choosing among $\approx 29{,}000$ tokens at every step.}
      \vspace{0.5em}

      {\small The gradient term spreads the logits by $\approx 0.009$ nats. The softmax that
      consumes them has a $10.82$-nat budget, so the gradient moves the proposal by about one
      part in a thousand.}
      \vspace{0.4em}

      \structure{The calibrated proposal is numerically uniform.}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{center}\small
  So a $5\times5$ sweep over step size and temperature re-tests everything below where the gradient \emph{does} shape the proposal.
  \end{center}
\end{frame}

\begin{frame}{The central result: is the \emph{direction} worth anything?}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \structure{The control.} Take the true gradient, throw away its direction, keep its length
      exactly. That is the \emph{norm-matched random} arm. The two differ in direction and in
      nothing else.
      \vspace{0.5em}

      \structure{Read the whole chain, not its end.} Three summaries of the same run:
      \begin{itemize}\small
        \item \textbf{last step}: where it stopped.
        \item \textbf{chain mean}: its typical state.
        \item \textbf{best state}: the best it ever reached.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Policy minus norm-matched random, $n=1000$ paired}
        \scriptsize
        \begin{tabular}{lrl}
          last step   & $+0.133$ & $[-0.063,\,+0.326]$ \\
          chain mean  & $+0.136$ & $[-0.012,\,+0.287]$ \\
          best state  & $+0.033$ & $[-0.079,\,+0.145]$ \\
        \end{tabular}
      \end{block}
      \vspace{-0.2em}
      {\small \structure{Margin $\pm0.327$ nats}, written down \emph{before} the test: any gap
      smaller than this is too small to matter. All three intervals fit inside it.}
      \vspace{0.4em}

      {\small So this is \structure{certified equivalence}, not ``we failed to find a
      difference''.}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{center}\small
  Same picture on all five energies, and in the sweep recovery never exceeds $2\%$ in any of the $25$ cells.
  \end{center}
\end{frame}

\section{Eliminating the explanations}

\begin{frame}{Hypothesis A: maybe the energy itself is hopeless?}
  \vspace{0.2em}
  {\small Then try something simpler: search the \emph{same} energy without ever differentiating it.}
  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.63\textwidth}
      \scriptsize
      \begin{tabular}{lcc}
        \toprule
        Method (no gradient of the energy) & KL & exact \\
        \midrule
        Ground-truth fill (floor) & $0.00$ & --- \\
        Corrupted token left in place & $9.14$ & $0\%$ \\
        Best Langevin cell \emph{(uses the gradient)} & $\sim6.4$ & $0.0\%$ \\
        Gradient-free Gibbs sampler & $6.69$ & $18.5\%$ \\
        Top-$k$ rescore, one forward pass & $\mathbf{4.43}$ & $\mathbf{33\%}$ \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.37\textwidth}
      \begin{block}{}
      \scriptsize \structure{Gibbs}: blank one position, ask the model for its distribution
      there, draw a token, move on. Repeat. No gradient anywhere.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{center}
  \structure{A is rejected: the energy is searchable, just not by this derivative.}\\
  {\footnotesize And the cheap methods are the good ones: no backward pass at all.}
  \end{center}
\end{frame}

\begin{frame}{Why, part 1: the approximation dies before the first step}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \small
      The proposal ranks a candidate by a \structure{straight-line estimate} of what swapping it in would do:
      \[ \hat{\Delta}(v) = \bm{g}^\top\!\big(\bm{e}(v)-\bm{e}(x_i)\big) \]
      A straight line only fits a curve \emph{near} the point it is drawn at. So: how far does it stay accurate?
      \vspace{0.4em}

      \structure{Measured:} correlate the estimate against the truth, \emph{within bins of distance}.
    \end{column}
    \begin{column}{0.56\textwidth}
      \begin{center}
      \includegraphics[width=0.86\textwidth]{fig_lin_radius.pdf}
      \\[0.1em]{\scriptsize Correlation with the true change, by how far the swap moves you.}
      \end{center}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{center}\small
  It is at zero by distance $\mathbf{1.82}$, which is the average gap between \emph{neighbouring}
  tokens: the \structure{smallest move that changes any word at all}.\\[0.1em]
  \structure{Every legal move lies outside the range where the estimate works.}
  \end{center}
\end{frame}

\begin{frame}{RQ2: what does the accept/reject step actually do?}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1cm, y=1cm, font=\tiny]

        \node[imsteal, font=\tiny, anchor=south] at (1.8,1.28) {forward};
        \draw[-{Latex[length=2mm]}, imsred, very thick, dashed] (2.62,1.19) -- (3.5,0.35);
        \node[imsred, font=\tiny, anchor=west] at (3.0,0.72) {reverse drift};
        \node[imsred, font=\tiny, align=center, anchor=north] at (1.9,0.15)

\section{Amortization and control}

\begin{frame}{GFlowNet amortization does not escape the failure}
  \vspace{0.3em}
  {\small A GFlowNet only ever \emph{evaluates} the reward, never differentiates it.}
  \vspace{0.8em}
  \begin{itemize}\small
    \item Three failure modes: \textbf{capacity collapse}, \textbf{length collapse}, \textbf{reward hacking}.
    \item Tuning moved the energy by $31$--$57$ nats\ldots
    \item \ldots yet Langevin sampling on the tuned energy is \emph{indistinguishable} from the untuned base.
  \end{itemize}

  \structure{Changing the model through this objective did not make the local surrogate useful.}

\section{Takeaways}

\begin{frame}{Takeaways: the four questions, answered}
  \vspace{0.2em}
  \scriptsize
  \setlength{\tabcolsep}{5pt}
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{p{2.5cm}p{8.1cm}}
    \toprule
    \structure{RQ1} \textbf{direction}
      & \textbf{No.} Same length, random direction: indistinguishable, inside a margin fixed in
        advance, on $1000$ paired sequences. The same frozen weights differentiated at the
        token's \emph{identity}: $0\% \to 40\%$. \\
    \structure{RQ2} \textbf{correction}
      & \textbf{Opposite things in the two samplers.} In the discrete one it supplies all of the
        selection pressure. In the continuous one it disables the sampler: a good move
        ($e^{+4.60}$) is rejected because the way back is unproposable ($e^{-1325}$). \\
    \structure{RQ3} \textbf{amortization}
      & \textbf{No escape.} The policy reaches high reward by exploiting it, and tuning moved the
        energy by $31$ to $57$ nats without making the local surrogate any more useful. \\
    \structure{E} \textbf{control}
      & \textbf{Only where the proposal can move.} Nothing on the frozen autoregressive
        landscape; on a bidirectional one the target moves, bounded by how far the guiding and
        judging instruments agree. \\
    \bottomrule
  \end{tabular}
  \vspace{0.5em}
  \begin{center}\small
  \structure{One thread through all four:} what a sampler can \emph{condition on} and \emph{read},
  not how hard it searches.
  \end{center}
\end{frame}

\begin{frame}{What it comes down to}
  \vspace{1.0em}
  \begin{center}
  \Large\structure{The signal was in the model all along.}\\[0.3em]
  \Large\structure{The derivative threw it away.}
  \end{center}
  \vspace{1.2em}
  \begin{itemize}\small
    \item Relax the token \emph{indicator} instead of the input embedding, or read the proposal
      off the model's output side. Same frozen weights, $0\% \to 40\%$.
    \item Let the proposal see the right context as well, and it goes further:
      RoBERTa $44.5\%$, never trained on scores.
  \end{itemize}
  \vspace{1.0em}
  \begin{center}
  {\footnotesize\color{gray!80} Scope: the null is about an input-embedding gradient of a
  likelihood whose target token enters as a discrete index. MuCoLa and COLD keep that self term,
  and are not refuted by it.}
  \end{center}
\end{frame}

\begin{frame}[noframenumbering]{Backup: SEDD and RoBERTa, where the proposal is read and where it enters}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \structure{Where the proposal is read off}
      \begin{itemize}\scriptsize
        \item \textbf{SEDD} (Lou et al.\ 2024), absorbing graph, dim $50{,}258 = 50{,}257 + $ MASK,
          GPT-2 tokenizer so no retokenization. The raw output $f_\theta(\bm{x},\sigma)$ is
          \emph{already} a log-space score.
        \item Read-out: set position $m$ to MASK, one forward at $\sigma = 0.1$, drop the MASK
          column, log-softmax over the real vocabulary $\Rightarrow \log q(v)$.
        \item \textbf{RoBERTa-large}, an ordinary MLM, never score-trained: put
          \texttt{<mask>} at $m$, one forward, log-softmax of the MLM head.
        \item Tokenizer bridge: RoBERTa id $\to$ GPT-2 id for every token whose surface
          re-encodes to exactly one GPT-2 token, $99.95\%$ coverage, restricted and renormalized.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \structure{Where it enters the sampler}
      \vspace{0.2em}
      \begin{center}
        {\scriptsize the proposal logit $t_2[v]$, and nothing else}\\[0.2em]
        $\tfrac12\,\bm{g}^\top\!\big(\bm{e}(v)-\bm{s}\big)$\\[0.15em]
        {\scriptsize\textcolor{imsred}{$\Downarrow$\ the one thing swapped}}\\[0.15em]
        $\tfrac12 \log q\big(v \mid \bm{x}_{\setminus m}\big)$
      \end{center}
      \vspace{0.2em}
      \begin{itemize}\scriptsize
        \item The accept step is \emph{untouched}:
          $\alpha = \min\{1,\ \tfrac{p(\bm{x}')\,q(\bm{x})}{p(\bm{x})\,q(\bm{x}')}\}$ with $p$ the
          exact frozen GPT-2 Large sequence likelihood.
        \item $q$ does not depend on the current token, so this is an \structure{independence
          sampler} and the Hastings ratio is closed form.
        \item Same sequences, same per-index corruption seeds, same energy, same KL metric. One
          variable moves.
      \end{itemize}
      \vspace{0.2em}
      {\scriptsize Exact recovery: gradient $0.0$ $\to$ left-conditional $23.5$ $\to$ SEDD
      $38.5 / 39.0$ $\to$ RoBERTa $\bm{44.5}$; uniform through the same code $0.5$.}

\begin{frame}[noframenumbering]{Backup: the Metropolis--Hastings chain}
  \small
  \[ \alpha = \min\!\left\{1,\ \frac{p(\bm{s}')\,q(\bm{s}\mid\bm{s}')}{p(\bm{s})\,q(\bm{s}'\mid\bm{s})}\right\} \]
  \begin{itemize}
    \item Target ratio: prefers good states. Proposal ratio: enforces reversibility.
    \item MALA needs a Lipschitz drift (Roberts and Tweedie 1996; Dalalyan 2017; Durmus and Moulines 2017).
    \item The projected target is piecewise constant; the differentiable pathway's drift jumps across a cell boundary, so the reverse density vanishes.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: what the sampler does without the correction}
  \small
  \begin{itemize}
    \item Correction off: a uniform random walk that never localizes. Mean KL \emph{rises} from $8.765$ to $9.499$; the chain still changes its token on $99.1\%$ of final-decile steps.
    \item So the late entropy dip is the annealing floor, not a quench: no freezing, no convergence to a local optimum.
    \item Correction on: the same uniform proposal becomes an independence sampler, the accept/reject filter does all the work, final KL $6.541$ at a $9.98\%$ acceptance rate.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: likelihood is a poor objective, yet used everywhere}
  \small
  \begin{itemize}
    \item The \textbf{likelihood trap}: the lowest-energy text is the most degenerate (Holtzman 2020, reproduced here). Within-strategy repetition correlation up to $+0.91$.
    \item Resolution: likelihood ranks single-token substitutions well (used by the KL metric and Gibbs), but maximizing absolute sequence likelihood in free generation is degenerate.
    \item So using it as the KL metric, the reward, and the SEDD comparison is consistent: relative-to-reference fit, not absolute likelihood as quality.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: MuCoLa / COLD, and why they appear to work}
  \small
  \begin{itemize}
    \item The continuous sampler follows their state \emph{geometry} faithfully, a continuous embedding state with a projection after every update, and adds the correction those works omit. It does \emph{not} follow their \emph{energy}: see backup 33.
    \item Without the exact correction the update, \emph{as reimplemented and measured here}, behaves as an early-stopped biased optimizer rather than a sampler from the stated energy. That is a statement about the mechanism, not a reinterpretation of those systems' published results, whose tasks, tuning and evaluation this study did not reproduce.
    \item The constrained \emph{mucola} arm is that comparison, run at the energy level (final KL under the same base model), not the task level.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: task design and corpus}
  \small
  \begin{itemize}
    \item Masked-token recovery on WikiText-2 validation; the base GPT-2 Large is fine-tuned on ROCStories (out-of-domain test).
    \item In-domain ROCStories diagnostics reproduce the null and the likelihood trap: not a domain-shift artifact.
    \item The KL metric compares the recovered token's next-token conditional against the ground-truth-conditioned reference; synonyms score well.
    \item Llama-3 is a cross-architecture control, never on a shared numerical axis. One Llama contrast is nominally significant against the gradient, but it is one of dozens uncorrected, is measured with the correction off, and does not reproduce on GPT-2 sharp cells; the supported claim is \emph{indifference}, not anti-guidance.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: why did the temperature flatten everything?}
  \small
  \begin{itemize}
    \item Proposal logit $=\ \big[\underbrace{\tfrac12 \bm{g}^\top(\bm{e}(v)-\bm{e}(x_i))}_{t_2} - \underbrace{\tfrac{1}{2\alpha}\lVert \bm{e}(v)-\bm{e}(x_i)\rVert^2}_{t_1}\big] / T$.
    \item At the calibrated cell $T=5$ divides a spread of a few nats into a $10.82$-nat entropy budget: the softmax is uniform whatever the surrogate says.
    \item Sharpness from the \emph{distance} term is sharp about \emph{staying put} ($t_2/t_1 = 1.8$); sharpness from the \emph{surrogate} is sharp about the ranking ($t_2/t_1 = 229$). Only the second tests the direction.
    \item The recovering cell needs large $\epsilon$ (kills $t_1$) \emph{and} low $T$ together.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: what exactly is the token-indicator derivative?}
  \small
  Relax the token indicator $\bm{z}_i$ from its simplex vertex, in \emph{both} roles:
  \[ \tilde{L}(\bm{z}_i) = \sum_v \bm{z}_i[v]\log p(v\mid \bm{x}_{<i}) \;+\; L_{\text{future}}(\bm{E}^\top \bm{z}_i) \]
  \begin{itemize}
    \item First term: the current position's own log-likelihood with the discrete \emph{target} relaxed. Second term: the suffix, with the relaxed indicator mapped through $\bm{E}$ to an input embedding.
    \item At a vertex $\tilde{L}$ equals the sequence log-likelihood, so this is a relaxation of the same object.
    \item It is \emph{not} $\bm{E}^\top\bm{g}$, the ordinary one-hot input gradient through the embedding lookup: that carries the future term only. The whole difference is the self term.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: why does RoBERTa beat SEDD?}
  \small
  \begin{itemize}
    \item Both are bidirectional; only SEDD is score-trained. RoBERTa $44.5\%$ vs SEDD-medium $39.0\%$, so the score objective is not what buys recovery.
    \item RoBERTa-large is bigger and trained on more text than SEDD-small/medium, and masked-token prediction \emph{is} its native query at any noise level, whereas SEDD answers it at a schedule point.
    \item So the claim is an \emph{ordering by conditioning}, not an isolation of it and not a ranking of the two models: the arms differ in scale, corpus and architecture as well, and only the tokenizer and the chain are held fixed across all of them. The SEDD arms are pilots.
    \item Tokenizer bridge: $99.95\%$ of RoBERTa ids map to one GPT-2 token; no ground-truth token fell outside it. Uniform control through the same code: $0.5\%$, so the pipeline leaks nothing.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Backup: why a GFlowNet, and the task-comparability caveat}
  \small
  \begin{itemize}
    \item A GFlowNet samples in proportion to reward (not the mode), and never differentiates it, so a gradient pathology would be invisible to it.
    \item It reformulates infilling as left-to-right generation under a restructured prompt; the samplers do in-place bidirectional substitution.
    \item The comparison is therefore run at the energy level, not the task level; the left-to-right reformulation is not adopted for the samplers because it would dissolve the revision capability under study.
  \end{itemize}
\end{frame}


\begin{frame}[noframenumbering]{Backup: how this differs from MuCoLa and COLD}
  \scriptsize
  \setlength{\tabcolsep}{4pt}
  \begin{center}
  \begin{tabular}{llc}
    \toprule
    Method & how the token's \emph{own} score enters & self term \\
    \midrule
    MuCoLa & softmax numerator $h_n^\top \tilde{\bm{e}}_{n+1}$, target relaxed & \textcolor{imsteal}{kept} \\
    COLD & $\sum_v p_{\text{LM}}(v\mid\tilde{y}_{<t})\log\mathrm{softmax}(\tilde{y}_t(v))$ & \textcolor{imsteal}{kept} \\
    CLS, DLS (this thesis) & gather at a hard, projected index & \textcolor{imsred}{discarded} \\
    token indicator (ours) & $\log p(v\mid \bm{x}_{<i})$, indicator relaxed & \textcolor{imsteal}{kept} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4em}
  \begin{itemize}
    \item \structure{No, they do not share the blindness.} Both relax the \emph{target} as well as the
      input, which is exactly the design that makes the self term differentiable. MuCoLa says so
      explicitly: $\tilde{\bm{e}}_{n+1}$ receives gradients ``directly from $-\log P$'' as well as
      through the later hidden states.
    \item \structure{The blind object is mine.} This implementation scores the masked position by
      gathering at the projected token id, so $\nabla_{\bm{e}_i}$ reaches the suffix only. The null is
      about \emph{that} energy in embedding coordinates.
    \item \structure{And the repair converges with theirs.} MuCoLa's self-gradient is $h_n$, ranking
      candidates by $h_n^\top(\bm{e}(v)-\bm{e}(x_i))$, the logit difference. The token-indicator self
      term is $\log p(v\mid\bm{x}_{<i}) - \log p(x_i\mid\bm{x}_{<i})$. Those agree up to the shared
      denominator, which cancels. The term worth $0 \to 40\%$ is the one MuCoLa already had.
    \item \structure{What was reimplemented faithfully} is their sampler \emph{geometry}: continuous
      embedding state, nearest-neighbour projection, plus the correction they omit. Not their energy.
  \end{itemize}
\end{frame}

  \begin{itemize}\small
    \item The apparent ordering is \structure{not} a separation: $\pm 12.5$ point intervals that
      overlap heavily. The better powered paired test puts both differences on zero: onehot minus
      self $+0.138$ $[-0.186,+0.449]$, embgrad minus self $+0.034$ $[-0.255,+0.320]$.
    \item So Q1's answer does not change when the task changes from repairing one token to
      conjuring twenty: the self term matches the full surrogate at $6.5\times$ less compute.
    \item \structure{One result the other way}: embgrad reaches $+12.52$ nats higher final LM
      log-probability, CI excluding zero. Not directional signal: all arms share the identical
      exact accept step and target the identical distribution, so this is \emph{mixing}.
  \end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Post-submission: three defects caught before any real run}
  \small
  Each would have produced a confident and wrong answer. All three were found by smoke tests.
  \begin{enumerate}\small
    \item \structure{Shortlist excludes the incumbent.} The exact arm put zero mass on the token
      the chain currently held, so the reverse probability was zero, $\log\alpha$ was $-\infty$,
      acceptance $0.0\%$ on every sequence.
    \item \structure{The distance term pins a shortlisted chain.} With $k=64$ and $t_1$ present,
      all five arms returned an identical final KL of $6.777$: $t_1$ scores the incumbent at $0$
      and every rival negatively. Over $50$k candidates that is one preference among many; over
      $65$ it is decisive.
    \item \structure{A state-dependent shortlist is not reversible.} Recomputing ``top-$k$ here
      plus where I am'' gives the reverse move probability zero as soon as the chain moves.
      Acceptance $0.0\%$ while the proposal was healthy: $24.6\%$ of proposal mass on the ground
      truth, the move worth $+19.8$ nats, refused because the reverse term was $-\infty$. Fixed
      by freezing the support per sequence.
  \end{enumerate}
  \vspace{0.2em}
  {\footnotesize Regression gate after every sampler edit: the flagship archived configuration
  reproduces bit-identically, maximum absolute difference $0.000\mathrm{e}{+}00$ on all three
  recorded channels.}
\end{frame}

\begin{frame}[noframenumbering]{Post-submission: five sentences}
  \small
  \begin{enumerate}\small
    \item The constructive half of the thesis \structure{needs no gradient at all}: the self term,
      which every autoregressive model already exposes, gets the whole effect at a quarter of the
      cost and zero backward passes.
    \item The $40$ percent headline is an $n=50$ cell that reproduces exactly; at $n=200$ it is

    \item Gradient-based constraint guidance is weak here \structure{for a measurable reason}: the
      classifier's first-order expansion has rank correlation near zero with the change it is
      expanding, and a likelihood-ranked shortlist almost never contains the token the constraint
      wants.
  \end{enumerate}
