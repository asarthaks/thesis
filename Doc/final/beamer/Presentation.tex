\section{Motivation}

\begin{frame}{The problem autoregressive models cannot solve natively}
  \begin{itemize}
    \item Autoregressive generation: one token at a time, left to right.
    \item A token is fixed the moment it is emitted.\\
          \emph{No mechanism to revise an earlier token in light of a later one.}
    \item Global properties (sentiment, factual constraint) live in the \emph{whole} sequence.
  \end{itemize}
  \vspace{0.6em}
  \begin{center}
  \begin{tikzpicture}[font=\footnotesize]
    \foreach \i/\w in {0/The, 1/movie, 2/was, 3/absolutely} {
      \node[box, minimum width=1.4cm] (t\i) at (\i*1.7,0) {\w};
    }
    \node[draw=imsred, thick, rounded corners, minimum width=1.4cm, fill=imsred!10] (t4) at (4*1.7,0) {?};
    \foreach \i in {0,...,3}{ \pgfmathtruncatemacro\j{\i+1} \draw[ar] (t\i) -- (t\j); }
    \draw[imsred, thick, -{Latex}, dashed] (t4.north) to[bend right=35] node[above, font=\scriptsize]{cannot flow back} (t0.north);
  \end{tikzpicture}
  \end{center}
  \begin{center}\small Control means steering the \emph{completed} object. That is a global problem.\end{center}
\end{frame}

\begin{frame}{The energy-based promise, and its one assumption}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      Treat the frozen likelihood as an \textbf{energy} over sequences and sample from it:
      \[ p(\bm{x}) \propto \exp\!\big(-E(\bm{x})\big),\quad E = -\log p_{\text{LM}} + \lambda\,C. \]
      \begin{itemize}
        \item Add any constraint $C$ at inference time.
        \item One frozen model, any objective. \textbf{Plug and play.}
      \end{itemize}
      \vspace{0.4em}
      \structure{The untested assumption:}\\
      \emph{the gradient of the frozen likelihood is a usable search direction on discrete text.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.9]
        \draw[->,gray] (-0.2,0) -- (4.4,0) node[right,font=\scriptsize]{sequence};

      \end{center}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{The machinery: Langevin dynamics}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      {\small Sample $p(\bm{s}) \propto e^{-E(\bm{s})}$ with a gradient-guided chain:}
      \[ \bm{s}_{t+1} = \bm{s}_t + \tfrac{\epsilon_t}{2}\,\nabla_{\bm{s}}\log p(\bm{s}_t) + \sqrt{\epsilon_t}\,\bm{\xi}_t \]
      \vspace{-1.0em}
      \begin{itemize}\small
        \item \textbf{Drift}: uphill.
        \item \textbf{Diffusion}: a \emph{sample}, not a mode.
        \item Needs a \textbf{Lipschitz} drift.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1cm, y=1cm, font=\scriptsize]
        \draw[imsblue!25, thick] plot[domain=0:4.3, samples=60] (\x, {0.55*sin(1.6*\x r)+0.55});
        \foreach \i/\xp in {1/0.35, 2/1.15, 3/1.95, 4/2.75, 5/3.6}
          \fill[imsblue] (\xp, {0.55*sin(1.6*\xp r)+0.55}) circle (1.6pt);
        \foreach \a/\bb in {0.35/1.15, 1.15/1.95, 1.95/2.75, 2.75/3.6}
          \draw[-{Latex[length=1.5mm]}, imsblue, thick]
            (\a, {0.55*sin(1.6*\a r)+0.55}) -- (\bb, {0.55*sin(1.6*\bb r)+0.55});
        \node[imsred, anchor=west, font=\tiny] at (0.4,1.75) {small steps, smooth surface};
      \end{tikzpicture}
      \end{center}
      \vspace{0.5em}
      {\small Text is \emph{discrete}: there is no small step.}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{center}\small
  Every method here \structure{forces this update onto discrete tokens}.
  \end{center}
\end{frame}

\begin{frame}{Research questions}
  \vspace{0.5em}
  \begin{enumerate}\small
    \item[RQ1] Does the \emph{input-embedding} gradient give a usable proposal direction? If not, what does?
    \item[RQ2] What does the Metropolis--Hastings correction do in each sampler?
    \item[RQ3a] Does a GFlowNet on this reward learn a high-reward policy?
    \item[RQ3b] Does its tuned energy become more navigable?
    \item[E] (extension) Can an additive constraint steer generation here?
  \end{enumerate}

\section{The faithful implementation}

\begin{frame}{Two samplers, implemented faithfully}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\small
        \item \textbf{DLS} (discrete): stays in token space.
        \item \textbf{CLS} (continuous): relaxes into embedding space, projects back.
        \item \textbf{Metropolis--Hastings} correction, applied exactly.
        \item Step size by \textbf{oracle calibration}.
      \end{itemize}
      \vspace{0.3em}
      {\scriptsize DLS scores a candidate $v$ against $x_i$, with $\bm{g} = \nabla_{\bm{e}_i}\log p_{\text{LM}}$:}
      {\footnotesize
      \begin{align*}
        \log q(v \mid x_i) \propto\;\; & \underbrace{\tfrac{1}{2}\bm{g}^\top\!\big(\bm{e}(v)-\bm{e}(x_i)\big)}_{\text{gradient alignment}} \\[0.2em]
        -\;\; & \underbrace{\tfrac{1}{2\alpha}\big\lVert \bm{e}(v)-\bm{e}(x_i)\big\rVert^2}_{\text{distance}}
      \end{align*}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[node distance=6mm and 9mm]
        \node[box] (s) {state};
        \node[op, right=of s] (g) {gradient\\$\to$ proposal};
        \node[op, below=of g] (mh) {MH\\accept / reject};
        \node[box, left=of mh] (p) {project\\to token};
        \draw[ar] (s) -- (g);
        \draw[ar] (g) -- (mh);
        \draw[ar] (mh) -- (p);
        \draw[ar] (p) -- (s);
        \node[font=\scriptsize, imsred, below=1.5mm of mh, text width=3.2cm, align=center] {the step the field skips};
      \end{tikzpicture}
      \\[0.4em]{\scriptsize A fixed annealed schedule, never stopped early.}
      \end{center}
      \vspace{0.8em}
      {\small This $\bm{g}$, the \structure{input-embedding gradient}, is the object under test.}
    \end{column}
  \end{columns}
\end{frame}

\section{The null}

\begin{frame}{First: what does the calibrated proposal actually look like?}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small Measure the entropy of the proposal the grid draws from.
      \begin{block}{Flagship configuration}
        \small
        step $0$: $10.8248$ nats\\
        floor over $50$ steps: $10.28$\\
        uniform ceiling $\log|V| = 10.8249$
      \end{block}
      {\small Gradient term: $\approx 0.009$ nats inside a $10.82$-nat budget.}
      \vspace{0.4em}

      \structure{The calibrated proposal is numerically uniform.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Why this comes first}
        \small
        Two proposals both within $10^{-4}$ nats of uniform are two draws from the \emph{same} distribution.
        \vspace{0.4em}

        \structure{A perfectly informative gradient would have produced the same null here.}
      \end{alertblock}
      {\small So a $5\times5$ sweep over step size and temperature re-tests it where the gradient \emph{does} shape the proposal.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The central result: gradient vs.\ a norm-matched random direction}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \small Isolate the gradient's \emph{direction} from its \emph{magnitude}.
      \begin{block}{Paired contrast, $n = 1000$}
        \scriptsize
        \begin{tabular}{lrl}
          last iterate  & $+0.133$ & $[-0.063,\,+0.326]$ \\
          chain mean    & $+0.136$ & $[-0.012,\,+0.287]$ \\
          chain minimum & $+0.033$ & $[-0.079,\,+0.145]$ \\
        \end{tabular}
      \end{block}
      {\small All inside the $\pm 0.327$ margin: \structure{equivalence certified}.}
      \vspace{0.4em}

      {\small In the sweep, recovery never exceeds $2\%$ in any of the $25$ cells.}
    \end{column}
    \begin{column}{0.44\textwidth}
      \scriptsize
      \begin{tabular}{lcc}
        \toprule
        Energy & policy & random \\
        \midrule
        GPT-2 Large & $6.474$ & $6.430$ \\
        GFN lb0-500 & $6.009$ & $6.001$ \\
        GFN lb0-2000 & $6.698$ & $6.552$ \\
        GFN lb1-500 & $6.008$ & $5.338$ \\
        \bottomrule
      \end{tabular}
      \\[0.5em]{\scriptsize Final KL, lower better.}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{center}
  \structure{The gradient direction is equivalent to noise of the same norm.}
  \end{center}
\end{frame}

\section{Eliminating the explanations}

\begin{frame}{Hypothesis A: is the target at fault?}
  \vspace{0.3em}
  {\small Score the \emph{same} energy without ever taking its gradient:}
  \vspace{0.7em}
  \begin{center}
  \small
  \begin{tabular}{lc}
    \toprule
    Method (no gradient of the energy) & mean KL \\
    \midrule
    Ground-truth fill (floor) & $0.00$ \\
    Corrupted token left in place & $9.14$ \\
    Best Langevin configuration \emph{(uses the gradient)} & $\sim 6.4$ \\
    Gradient-free Gibbs sampler & $6.69$ \\
    Top-$k$ rescore, one forward pass & $\mathbf{4.43}$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.8em}
  \begin{center}
  \structure{A is rejected: the energy is searchable, just not by this derivative.}
  \end{center}
\end{frame}

\begin{frame}{Why, part 1: the linearization fails before the first move}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      \structure{Linearization} means replacing the true change a substitution causes by its first-order Taylor term, which is exactly what the proposal ranks by:
      \[ \hat{\Delta}(v) = \bm{g}^\top\!\big(\bm{e}(v)-\bm{e}(x_i)\big) \]
      \[ \Delta_{\text{true}}(v) = \log p(\bm{x}_{i\to v}) - \log p(\bm{x}) \]
      Correlate the two over $n = 400{,}000$ candidate--sequence pairs.
      \vspace{0.3em}

      $|\rho| < 0.06$ on \emph{all five} energies.
      \vspace{0.3em}

      \structure{The gradient is inapplicable, not merely imprecise.}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{center}
      \begin{tikzpicture}[scale=0.85, font=\scriptsize]
        \foreach \x/\y in {0/0, 1.6/0.3, 3.0/-0.2, 1.0/1.4, 2.4/1.2, 0.4/-1.2, 2.9/-1.4}{
          \fill[imsblue] (\x,\y) circle (1.2pt);
        }
        \node[circle, draw=imsteal, thick, minimum size=0.5cm] (r) at (0,0) {};

        \node[align=center, font=\scriptsize, text width=4.6cm] at (1.4,-2.0) {valid radius $\ll$ smallest admissible substitution};
      \end{tikzpicture}
      \\[0.2em]
      \includegraphics[width=0.84\textwidth]{fig_lin_radius.pdf}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why, part 2: the correction breaks the continuous sampler}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Decompose the acceptance ratio for boundary-crossing moves:
      \begin{block}{The two terms}
        \small
        target log-ratio $= +4.60$ \structure{(good moves)}\\
        proposal log-ratio $= -1325$ \structure{(kills them)}

      \begin{itemize}
        \item The projected drift is not Lipschitz across a cell boundary.
        \item CLS accepts within-cell moves $\approx 0\%$; DLS, $100\%$.
      \end{itemize}
      \structure{The exact correction disables CLS; it rescues DLS.}\\
      {\scriptsize Exact for the proposal it corrects. What fails is the regularity underneath.}
    \end{column}

\end{frame}

\section{Amortization and control}

\begin{frame}{GFlowNet amortization does not escape the failure}
  \vspace{0.3em}
  {\small A GFlowNet only ever \emph{evaluates} the reward, never differentiates it.}
  \vspace{0.8em}
  \begin{itemize}\small
    \item Three failure modes: \textbf{capacity collapse}, \textbf{length collapse}, \textbf{reward hacking}.
    \item Tuning moved the energy by $31$--$57$ nats\ldots
    \item \ldots yet Langevin sampling on the tuned energy is \emph{indistinguishable} from the untuned base.
  \end{itemize}
  \vspace{0.9em}
  \begin{center}
  \structure{Changing the model through this objective did not make the local surrogate useful.}
  \end{center}
\end{frame}

\begin{frame}{The extension: does an additive constraint steer anything?}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small $E = -\log p_{\text{LM}} + \lambda\,C$, sentiment head on the frozen base.
      \vspace{0.2em}

      {\scriptsize Raw gains are swamped by a sentiment drift, so the statistic is a paired contrast that cancels it.}
      \vspace{0.4em}
      \scriptsize

      \structure{On a landscape a proposal can navigate:}
      \begin{itemize}\small
        \item Guidance moves the classifier's own verdict by $+9.3$ to $+25.7$ points.
        \item Transfer to an independent judge is bounded by instrument agreement, $56$--$80\%$.

      \end{itemize}

  \begin{center}\small
  \structure{A constraint added to a landscape no sampler can navigate is added to nothing.}
  \end{center}
\end{frame}

\section{Takeaways}

\begin{frame}{Takeaways}
  \vspace{0.4em}
  \begin{enumerate}\small
    \item \structure{Not the derivative you think.} The \emph{input-embedding} gradient is equivalent to a norm-matched random direction, certified over $1000$ paired sequences.
    \item \structure{Not the target, and not the model.} The same frozen weights, differentiated in token-indicator coordinates, recover $40\%$ and $41\%$.
    \item \structure{What buys the rest is right-context access}, not a score objective: RoBERTa $44.5\% >$ SEDD $39\% >$ AR conditional $23.5\%$.
    \item \structure{Control follows navigation.}
  \end{enumerate}
  \vspace{0.9em}
  \begin{center}
  \Large\structure{Differentiate the right object, or propose from the output side.}\\[0.3em]
  \normalsize Not: evaluate rather than differentiate. The derivative was in the wrong coordinates.
  \end{center}
\end{frame}

\begin{frame}{Backup: the linearization radius and self-term blindness}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}

      \begin{itemize}
        \item The surrogate correlates with the \emph{future} term ($\rho = 0.03$), never the \emph{self} term ($\rho = -0.10$).
        \item Mean absolute self term $15.0$ nats; future term $24.2$ nats.
        \item The input-embedding gradient is structurally blind to the candidate's own left-context fit, which enters as a discrete index.
      \end{itemize}

\begin{frame}{Backup: the Metropolis--Hastings chain}
  \small
  \[ \alpha = \min\!\left\{1,\ \frac{p(\bm{s}')\,q(\bm{s}\mid\bm{s}')}{p(\bm{s})\,q(\bm{s}'\mid\bm{s})}\right\} \]
  \begin{itemize}
    \item Target ratio: prefers good states. Proposal ratio: enforces reversibility.
    \item MALA needs a Lipschitz drift (Roberts and Tweedie 1996; Dalalyan 2017; Durmus and Moulines 2017).
    \item The projected target is piecewise constant; the differentiable pathway's drift jumps across a cell boundary, so the reverse density vanishes.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: what the sampler does without the correction}
  \small
  \begin{itemize}
    \item Correction off: a uniform random walk that never localizes. Mean KL \emph{rises} from $8.765$ to $9.499$; the chain still changes its token on $99.1\%$ of final-decile steps.
    \item So the late entropy dip is the annealing floor, not a quench: no freezing, no convergence to a local optimum.
    \item Correction on: the same uniform proposal becomes an independence sampler, the accept/reject filter does all the work, final KL $6.541$ at a $9.98\%$ acceptance rate.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: likelihood is a poor objective, yet used everywhere}
  \small
  \begin{itemize}
    \item The \textbf{likelihood trap}: the lowest-energy text is the most degenerate (Holtzman 2020, reproduced here). Within-strategy repetition correlation up to $+0.91$.
    \item Resolution: likelihood ranks single-token substitutions well (used by the KL metric and Gibbs), but maximizing absolute sequence likelihood in free generation is degenerate.
    \item So using it as the KL metric, the reward, and the SEDD comparison is consistent: relative-to-reference fit, not absolute likelihood as quality.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: MuCoLa / COLD, and why they appear to work}
  \small
  \begin{itemize}
    \item The continuous sampler follows their state \emph{geometry} faithfully, a continuous embedding state with a projection after every update, and adds the correction those works omit. It does \emph{not} follow their \emph{energy}: see backup 33.
    \item Without the exact correction the update, \emph{as reimplemented and measured here}, behaves as an early-stopped biased optimizer rather than a sampler from the stated energy. That is a statement about the mechanism, not a reinterpretation of those systems' published results, whose tasks, tuning and evaluation this study did not reproduce.
    \item The constrained \emph{mucola} arm is that comparison, run at the energy level (final KL under the same base model), not the task level.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: task design and corpus}
  \small
  \begin{itemize}
    \item Masked-token recovery on WikiText-2 validation; the base GPT-2 Large is fine-tuned on ROCStories (out-of-domain test).
    \item In-domain ROCStories diagnostics reproduce the null and the likelihood trap: not a domain-shift artifact.
    \item The KL metric compares the recovered token's next-token conditional against the ground-truth-conditioned reference; synonyms score well.
    \item Llama-3 is a cross-architecture control, never on a shared numerical axis. One Llama contrast is nominally significant against the gradient, but it is one of dozens uncorrected, is measured with the correction off, and does not reproduce on GPT-2 sharp cells; the supported claim is \emph{indifference}, not anti-guidance.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: why did the temperature flatten everything?}
  \small
  \begin{itemize}
    \item Proposal logit $=\ \big[\underbrace{\tfrac12 \bm{g}^\top(\bm{e}(v)-\bm{e}(x_i))}_{t_2} - \underbrace{\tfrac{1}{2\alpha}\lVert \bm{e}(v)-\bm{e}(x_i)\rVert^2}_{t_1}\big] / T$.
    \item At the calibrated cell $T=5$ divides a spread of a few nats into a $10.82$-nat entropy budget: the softmax is uniform whatever the surrogate says.
    \item Sharpness from the \emph{distance} term is sharp about \emph{staying put} ($t_2/t_1 = 1.8$); sharpness from the \emph{surrogate} is sharp about the ranking ($t_2/t_1 = 229$). Only the second tests the direction.
    \item The recovering cell needs large $\epsilon$ (kills $t_1$) \emph{and} low $T$ together.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: what exactly is the token-indicator derivative?}
  \small
  Relax the token indicator $\bm{z}_i$ from its simplex vertex, in \emph{both} roles:
  \[ \tilde{L}(\bm{z}_i) = \sum_v \bm{z}_i[v]\log p(v\mid \bm{x}_{<i}) \;+\; L_{\text{future}}(\bm{E}^\top \bm{z}_i) \]
  \begin{itemize}
    \item First term: the current position's own log-likelihood with the discrete \emph{target} relaxed. Second term: the suffix, with the relaxed indicator mapped through $\bm{E}$ to an input embedding.
    \item At a vertex $\tilde{L}$ equals the sequence log-likelihood, so this is a relaxation of the same object.
    \item It is \emph{not} $\bm{E}^\top\bm{g}$, the ordinary one-hot input gradient through the embedding lookup: that carries the future term only. The whole difference is the self term.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: why does RoBERTa beat SEDD?}
  \small
  \begin{itemize}
    \item Both are bidirectional; only SEDD is score-trained. RoBERTa $44.5\%$ vs SEDD-medium $39.0\%$, so the score objective is not what buys recovery.
    \item RoBERTa-large is bigger and trained on more text than SEDD-small/medium, and masked-token prediction \emph{is} its native query at any noise level, whereas SEDD answers it at a schedule point.
    \item So the claim is an \emph{ordering by conditioning}, not an isolation of it and not a ranking of the two models: the arms differ in scale, corpus and architecture as well, and only the tokenizer and the chain are held fixed across all of them. The SEDD arms are pilots.
    \item Tokenizer bridge: $99.95\%$ of RoBERTa ids map to one GPT-2 token; no ground-truth token fell outside it. Uniform control through the same code: $0.5\%$, so the pipeline leaks nothing.
  \end{itemize}
\end{frame}

\begin{frame}{Backup: why a GFlowNet, and the task-comparability caveat}
  \small
  \begin{itemize}
    \item A GFlowNet samples in proportion to reward (not the mode), and never differentiates it, so a gradient pathology would be invisible to it.
    \item It reformulates infilling as left-to-right generation under a restructured prompt; the samplers do in-place bidirectional substitution.
    \item The comparison is therefore run at the energy level, not the task level; the left-to-right reformulation is not adopted for the samplers because it would dissolve the revision capability under study.
  \end{itemize}
\end{frame}


\begin{frame}{Backup: how this differs from MuCoLa and COLD}
  \scriptsize
  \setlength{\tabcolsep}{4pt}
  \begin{center}
  \begin{tabular}{llc}
    \toprule
    Method & how the token's \emph{own} score enters & self term \\
    \midrule
    MuCoLa & softmax numerator $h_n^\top \tilde{\bm{e}}_{n+1}$, target relaxed & \textcolor{imsteal}{kept} \\
    COLD & $\sum_v p_{\text{LM}}(v\mid\tilde{y}_{<t})\log\mathrm{softmax}(\tilde{y}_t(v))$ & \textcolor{imsteal}{kept} \\
    CLS, DLS (this thesis) & gather at a hard, projected index & \textcolor{imsred}{discarded} \\
    token indicator (ours) & $\log p(v\mid \bm{x}_{<i})$, indicator relaxed & \textcolor{imsteal}{kept} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4em}
  \begin{itemize}
    \item \structure{No, they do not share the blindness.} Both relax the \emph{target} as well as the
      input, which is exactly the design that makes the self term differentiable. MuCoLa says so
      explicitly: $\tilde{\bm{e}}_{n+1}$ receives gradients ``directly from $-\log P$'' as well as
      through the later hidden states.
    \item \structure{The blind object is mine.} This implementation scores the masked position by
      gathering at the projected token id, so $\nabla_{\bm{e}_i}$ reaches the suffix only. The null is
      about \emph{that} energy in embedding coordinates.
    \item \structure{And the repair converges with theirs.} MuCoLa's self-gradient is $h_n$, ranking
      candidates by $h_n^\top(\bm{e}(v)-\bm{e}(x_i))$, the logit difference. The token-indicator self
      term is $\log p(v\mid\bm{x}_{<i}) - \log p(x_i\mid\bm{x}_{<i})$. Those agree up to the shared
      denominator, which cancels. The term worth $0 \to 40\%$ is the one MuCoLa already had.
    \item \structure{What was reimplemented faithfully} is their sampler \emph{geometry}: continuous
      embedding state, nearest-neighbour projection, plus the correction they omit. Not their energy.
  \end{itemize}
\end{frame}
